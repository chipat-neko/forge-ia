export const RARETE = (() => {
  // Le pari « contexte » et l'étape 5 : la part des contextes jamais vus, selon n.
  const f = faitsDuPariContexte(TEXTE_DU_SITE);
  const [entrainement, validation] = DECOUPE;
  const partDe = (n) => {
    const [absents, total] = partJamaisVue(compter(entrainement, n), validation, n);
    return absents / total;
  };
  const nA = 2;
  const nB = 4;
  const p2 = partDe(nA);
  const p4 = partDe(nB);
  const p6 = f.absents / f.total;
  exiger(p2 < p4 && p4 < p6, 'la part des contextes jamais vus ne croît plus avec n : l’étape 5 est à revoir');
  exiger(Math.round(1 / p6) === 3, `à n = ${f.n}, ${pourcent(p6)} des contextes sont jamais vus : « près d’un sur trois » est à revoir`);
  const possibles = f.v ** f.n;
  exiger(possibles > 1000 * f.differents, 'les contextes possibles ne sont plus mille fois plus nombreux que ceux du texte : l’étape 5 est à revoir');
  return {
    n: f.n,
    nA,
    nB,
    absents: nombre(f.absents),
    total: nombre(f.total),
    part: pourcent(p6),
    p2: pourcent(p2),
    p4: pourcent(p4),
    differents: nombre(f.differents),
    possibles: formaterQuantite(possibles),
    v: nombre(f.v),
  };
})();

export const PERPLEXITE = (() => {
  // L'étape 7 : le minimum de la perplexité de validation, et ce que font les deux courbes autour de lui.
  const { n, valeur } = minimumDesPerplexites(PERPLEXITES);
  const descendPuisMonte = (cle) =>
    PERPLEXITES.every((p, i) => i === 0 || (p.n <= n ? p[cle] < PERPLEXITES[i - 1][cle] : p[cle] > PERPLEXITES[i - 1][cle]));
  exiger(descendPuisMonte('validation'), `la perplexité de validation ne descend plus jusqu’à n = ${n} puis ne remonte plus à chaque caractère de plus`);
  exiger(descendPuisMonte('entrainement'), `la perplexité sur le texte d’entraînement n’a plus son minimum à n = ${n} : « remonte aussi » (étape 7) est à revoir`);
  const derniere = PERPLEXITES.at(-1);
  const de = (k) => nombre(PERPLEXITES.find((p) => p.n === k).validation, 1);
  // La rareté des longs contextes, qui fait remonter la perplexité même sur le texte d'entraînement.
  const longs = compter(DECOUPE[0], derniere.n);
  let rares = 0;
  for (const suivants of longs.values()) {
    let total = 0;
    for (const k of suivants.values()) total += k;
    if (total <= 2) rares += 1;
  }
  exiger(rares / longs.size > 0.5, 'les contextes de 6 caractères vus une ou deux fois ne sont plus la majorité : l’étape 7 est à revoir');
  // Le modèle qui écrit (comptes bruts, recul) donne une chance nulle aux caractères jamais vus après son
  // contexte : combien de caractères du texte de validation en reçoivent une, avec les réglages de l'atelier.
  const comptesEcriture = compterParOrdre(DECOUPE[0], N_INITIAL);
  const validation = Array.from(DECOUPE[1]);
  let nuls = 0;
  let positions = 0;
  for (let i = N_INITIAL; i < validation.length; i++) {
    positions += 1;
    let suivants;
    for (let k = N_INITIAL; k >= 0; k--) {
      suivants = comptesEcriture[k].get(validation.slice(i - k, i).join(''));
      if (suivants !== undefined) break;
    }
    if (!suivants.has(validation[i])) nuls += 1;
  }
  exiger(nuls > 0, 'le modèle qui écrit ne donne plus une chance nulle à aucun caractère du texte de validation : l’étape 7 est à revoir');
  const minimumEntrainement = PERPLEXITES.find((p) => p.n === n).entrainement;
  return {
    n,
    nCaracteres: caracteres(n),
    valeur: nombre(valeur, 1),
    nFin: derniere.n,
    fin: nombre(derniere.validation, 1),
    entrainementMin: nombre(minimumEntrainement, 1),
    entrainementFin: nombre(derniere.entrainement, 1),
    nuls: nombre(nuls),
    positions: nombre(positions),
    nEcriture: N_INITIAL,
    pari: NS_DU_PARI_PERPLEXITE.map((k) => ({ n: k, valeur: de(k) })),
    rares: pourcent(rares / longs.size),
    v: nombre(ALPHABET.length),
  };
})();

<Etape numero={1} titre="Prédire la suite">

Au <a href={TEXTE.lien01}>chapitre 0.1</a>, ton programme devinait le mot suivant : il comptait, dans un texte, quel mot suit chaque mot, puis prenait toujours le plus fréquent, sans hasard.

Un modèle de langage fait le même travail, autrement : au lieu d'un seul gagnant, il donne une probabilité à chaque suite possible. Pour écrire, il tire la suite au sort selon ces probabilités, l'ajoute au texte, et recommence. Les grands modèles qui répondent dans une conversation ne font rien d'autre, un morceau après l'autre.

Ce chapitre en construit un, le plus simple qui soit. Il travaille caractère par caractère, et ne regarde que les quelques caractères qui précèdent : leur groupe s'appelle le contexte. Pour apprendre, il lui faut un livre. Le sien, c'est ce site : la prose des chapitres 0.1 à 0.5 et 1.3, soit {TEXTE.caracteres} caractères. Il en lit les neuf premiers dixièmes, son texte d'entraînement ; l'étape 6 dira à quoi sert le reste.

<Etape numero={2} titre="Des probabilités, jamais zéro">

Le modèle commence par compter. Pour chaque contexte de n caractères du texte, il note quels caractères le suivent, et combien de fois :

<CodeExplique fichier="projets/1.1-modele-de-langage/modele_de_langage/ngrammes.py" region="compter">
  <li data-lignes="1">`compter` reçoit le texte et n, la longueur du contexte. Elle rend un dictionnaire : à chaque contexte, un `Counter` de ses suivants, comme au chapitre 0.1.</li>
  <li data-lignes="3">Avec n = 0, le contexte est le texte vide : on compte les caractères tout court.</li>
  <li data-lignes="5-6">`i` parcourt le texte, et `texte[i - n : i]`, une tranche, prend les n caractères qui précèdent le caractère `i`.</li>
  <li data-lignes="7-8">Un contexte encore jamais vu reçoit un `Counter` vide.</li>
  <li data-lignes="9">On ajoute 1 au compte du caractère qui suit ce contexte.</li>
</CodeExplique>

Avec n = {N_OUVERTURE}, le contexte est un seul caractère. Dans le texte d'entraînement, « {APRES_E.regarde} » est suivi {APRES_E.total} fois d'un caractère : {APRES_E.espace} fois d'une espace, {APRES_E.s} fois d'un « s », et ainsi de suite, {APRES_E.differents} caractères différents en tout.

Pour passer des comptes aux probabilités, il suffit de diviser par le total. Mais un caractère jamais vu après ce contexte aurait alors une probabilité de zéro : le modèle le jugerait impossible. À l'étape 7, on mesurera sa surprise par −ln de la chance qu'il donnait au bon caractère, comme l'erreur du <a href={TEXTE.lien02}>chapitre 0.2</a>. Et −ln 0 est infini : un seul zéro casserait la mesure. On ajoute donc 1 au compte de chacun des {TEXTE.v} caractères de l'alphabet du texte. C'est le lissage de Laplace :

<CodeExplique fichier="projets/1.1-modele-de-langage/modele_de_langage/ngrammes.py" region="distribution">
  <li data-lignes="1-6">`distribution` reçoit les comptes, l'alphabet, un contexte et n. Elle rend chaque caractère avec sa probabilité.</li>
  <li data-lignes="10-11">Les suivants des n derniers caractères du contexte. Un contexte jamais vu n'a aucun suivant : un `Counter` vide.</li>
  <li data-lignes="12-16">Le compte de chaque caractère, plus 1, divisé par le total des comptes plus la taille de l'alphabet. Les probabilités font bien 1 en tout.</li>
  <li data-lignes="19">Du plus probable au moins probable, comme `predire` au chapitre 0.1.</li>
</CodeExplique>

La chance de l'espace, dans l'ouverture, vient de là : <span class="formule-en-texte">({APRES_E.espace} + 1) ÷ ({APRES_E.total} + {TEXTE.v})</span>, soit {APRES_E.pEspace}.

<Pari
  id="suivant"
  question={`Avec le lissage, quelle chance le modèle donne-t-il à « ${SUIVANT.caractere} » après « ${SUIVANT.contexte} » ?`}
  precision={`Dans le texte d'entraînement, « ${SUIVANT.contexte} » apparaît ${SUIVANT.fois} fois, toujours suivi de « ${SUIVANT.caractere} ». L'alphabet du texte compte ${SUIVANT.v} caractères.`}
  choix={choixSuivant(TEXTE_DU_SITE)}
  acquis="distribution"
>

Environ une fois sur {SUIVANT.uneSur} : <span class="formule-en-texte">({SUIVANT.fois} + 1) ÷ ({SUIVANT.fois} + {SUIVANT.v})</span>, soit {SUIVANT.p}. Pourtant, « {SUIVANT.caractere} » suit « {SUIVANT.contexte} » {SUIVANT.fois} fois sur {SUIVANT.fois} : c'est le début de « ton IA », le nom de l'IA que tu fabriques. Mais chacun des {SUIVANT.autres} autres caractères reçoit aussi 1 chance sur {SUIVANT.denominateur}, soit {SUIVANT.pAutre}, et ensemble, ils prennent {SUIVANT.reste}. Aucun n'est impossible.

Quand un contexte est rare, le lissage pèse lourd : ici, {SUIVANT.fois} observations contre {SUIVANT.v} fois « +1 ». Pour mesurer, c'est le prix à payer pour éviter le zéro ; l'étape 7 montrera ce qu'il coûte. Pour écrire, un tirage dans cette distribution choisirait le plus souvent un caractère jamais vu à cet endroit : l'étape 3 tire donc autrement.

</Pari>

<Etape numero={3} titre="Tirer au sort">

Pour écrire, le modèle ne tire que parmi les caractères déjà vus après le contexte, chacun selon son compte : un caractère vu 30 fois sort 30 fois plus souvent qu'un caractère vu une fois. La fonction `poids` donne ces comptes, retouchés par la température du <a href={TEXTE.lien02}>chapitre 0.2</a> :

<CodeExplique fichier="projets/1.1-modele-de-langage/modele_de_langage/ngrammes.py" region="poids">
  <li data-lignes="1-5">`poids` reçoit les suivants d'un contexte, l'alphabet et la température. Elle rend un poids par caractère de l'alphabet.</li>
  <li data-lignes="7">Le compte de chaque caractère de l'alphabet après ce contexte : 0 pour un caractère jamais vu là.</li>
  <li data-lignes="8-9">À la température 1, le poids est le compte : les chances suivent ce que le modèle a lu. Un caractère jamais vu a un poids nul : il ne sort jamais.</li>
  <li data-lignes="10-11">À 0,5, le compte est élevé au carré : un caractère vu 3 fois pèse 9, un caractère vu une fois pèse toujours 1. Le plus fréquent sort encore plus souvent.</li>
  <li data-lignes="12-13">À 2, sa racine carrée : 3 devient environ 1,7. Les écarts se resserrent, et les caractères rares sortent plus souvent.</li>
  <li data-lignes="14-23">À 0, plus de hasard : seul le plus compté a un poids, de 1. À égalité, le premier dans l'ordre des numéros des caractères.</li>
</CodeExplique>

Reste un cas : un contexte que le modèle n'a jamais vu. Il n'a alors aucun compte. Il raccourcit son contexte : il regarde un caractère de moins, puis un autre, jusqu'à un groupe qu'il connaît. Au pire, il ne regarde plus rien, et tire selon la fréquence de chaque caractère dans tout le texte d'entraînement. Le tirage d'une suite rassemble tout cela :

<Etape numero={5} titre="Plus de contexte ?">

Plus le modèle regarde loin, plus ses suites ressemblent au texte du site. Pourquoi ne pas lui donner 10, 20 ou 100 caractères de contexte ? Pour le savoir, il faut un texte qu'il n'a jamais lu : le dernier dixième du texte du site, son texte de validation, a été mis de côté. L'étape 6 dira pourquoi.

<Pari
  id="contexte"
  question={`Dans le texte de validation, prends chaque groupe de ${RARETE.n} caractères. Quelle part de ces groupes le modèle n'a-t-il jamais vue dans le texte d'entraînement ?`}
  choix={choixContexte(TEXTE_DU_SITE)}
  acquis="contexte"
>

{RARETE.absents} groupes sur {RARETE.total}, soit {RARETE.part} : près d'un sur trois. Avec {RARETE.nA} caractères, la part n'est que de {RARETE.p2} ; avec {RARETE.nB}, de {RARETE.p4}.

Chaque caractère de plus multiplie par {RARETE.v} le nombre de contextes possibles. Avec {RARETE.n} caractères, il y en a {RARETE.possibles}, et le texte d'entraînement n'en contient que {RARETE.differents} différents. Devant un contexte qu'il n'a jamais vu, le modèle ne sait rien, même si ce contexte ressemble beaucoup à un contexte connu : il ne peut que raccourcir son regard.

</Pari>

En 2003, Yoshua Bengio et trois collègues ont décrit ce mur, qu'on appelle la malédiction de la dimension : plus le contexte est long, plus le texte à prédire contient des suites jamais vues pendant l'entraînement. Leur réponse : un réseau de neurones qui représente chaque mot par une liste de nombres, pour que des contextes qui se ressemblent partagent ce que le modèle en sait. Ton IA prendra ce chemin.

<Fragment slot="note">

Yoshua Bengio, Réjean Ducharme, Pascal Vincent et Christian Jauvin, [« A Neural Probabilistic Language Model »](https://www.jmlr.org/papers/v3/bengio03a.html), Journal of Machine Learning Research, 2003 (vérifié le 9 novembre 2026).

Ton IA, dans sa plus grande taille, regarde jusqu'à {LABO.tonIA} jetons en arrière, pas {RARETE.n} caractères. Son réseau ne compte pas les contextes : il apprend à rapprocher ceux qui se ressemblent.

</Fragment>

</Etape>

<Etape numero={6} titre="Le test du texte jamais lu">

Pour juger un modèle, on ne l'interroge pas sur ce qu'il a lu : il pourrait l'avoir retenu par cœur. La fonction `couper` du projet sépare donc le texte du site en deux. Les 90 premiers pour cent, {TEXTE.entrainement} caractères, servent à compter : c'est le texte d'entraînement. Les {TEXTE.validation} derniers, jamais comptés, servent à vérifier : c'est le texte de validation.

Tout ce chapitre suit cette règle : l'ouverture, les paris et l'atelier ne se servent que des comptes du texte d'entraînement. Le pari de l'étape 5 et la mesure de l'étape 7, eux, portent sur le texte de validation.

<Fragment slot="note">

La coupe se fait à <code class="insecable">len(texte) * 9 // 10</code> : `//` est la division entière de Python, qui tombe toujours sur un nombre entier de caractères.

</Fragment>

</Etape>

<Etape numero={7} titre="La perplexité">

Reste à mesurer. Pour chaque caractère du texte de validation, on regarde la chance que le modèle lui donnait, avec le lissage de l'étape 2, et on en prend −ln : 0 si le modèle était sûr, un grand nombre s'il était surpris. On fait la moyenne, puis on en prend l'exponentielle. C'est la perplexité.

Elle se lit comme un nombre de choix : une perplexité de 12, c'est un modèle qui hésite autant que s'il choisissait au hasard entre 12 caractères. Un modèle qui ne saurait rien hésiterait entre les {PERPLEXITE.v}.

<CodeExplique fichier="projets/1.1-modele-de-langage/modele_de_langage/ngrammes.py" region="perplexite">
  <li data-lignes="1-6">`perplexite` reçoit les comptes, l'alphabet, n et le texte à mesurer.</li>
  <li data-lignes="13-17">Pour chaque caractère du texte, sauf les n premiers, sa probabilité lissée après les n caractères qui le précèdent : le calcul de `distribution`, pour un seul caractère.</li>
  <li data-lignes="18-19">On ajoute sa surprise, −ln p, et on compte les caractères.</li>
  <li data-lignes="22">L'exponentielle de la surprise moyenne. `math.exp` et `math.log` sont l'exponentielle et le logarithme naturel du chapitre 0.2.</li>
</CodeExplique>

<p data-test="minimum">Sur le texte de validation, la perplexité est la plus basse avec {PERPLEXITE.nCaracteres} de contexte : {PERPLEXITE.valeur}. Au-delà, elle remonte, jusqu'à {PERPLEXITE.fin} avec {PERPLEXITE.nFin} caractères. Sur le texte d'entraînement aussi, elle remonte après {PERPLEXITE.n} caractères, de {PERPLEXITE.entrainementMin} à {PERPLEXITE.entrainementFin}, alors que le modèle y a vu chaque contexte.</p>

Des deux côtés, la cause est la rareté des longs contextes, que le lissage fait payer. {PERPLEXITE.rares} des contextes de {PERPLEXITE.nFin} caractères du texte d'entraînement n'y apparaissent qu'une ou deux fois, et près d'un sur trois de ceux du texte de validation n'y figure jamais. Devant si peu de comptes, les {PERPLEXITE.v} fois « +1 » l'emportent : la chance du bon caractère s'effondre. La mesure ne se trompe pas : c'est ce modèle lissé, celui de l'étape 2, qui devine mal.

Le modèle qui écrit, celui de l'atelier, échappe à ce +1 : il ne tire que parmi les caractères vus, et recule devant un contexte inconnu. Mais il donne une chance nulle à tout caractère jamais vu après son contexte. Avec {PERPLEXITE.nEcriture} caractères, c'est le cas de {PERPLEXITE.nuls} caractères du texte de validation sur {PERPLEXITE.positions} : un seul suffirait à rendre −ln infini, et sa perplexité avec. Ce sont deux modèles différents. Le meilleur pour la perplexité, avec {MOTS.nMin} caractères, n'est donc pas celui dont les suites ressemblent le plus au texte : dans la suite du départ de l'atelier, {MOTS.avecN} des mots existent avec {MOTS.n} caractères de contexte, et {MOTS.avecMin} seulement avec {MOTS.nMin}. Aucun des deux ne sait deviner après un contexte jamais vu : compter ne suffit pas.

</Etape>

<Etape numero={8} titre="Ce que fera ton IA de plus">

Ton IA fera le même travail que ce modèle : donner une probabilité à chaque suite possible, tirer au sort, recommencer. Trois choses changent.

Elle lira des jetons, les morceaux du <a href={TEXTE.lien13}>chapitre 1.3</a>, plutôt que des caractères : un mot fréquent tient en un seul jeton, et le même contexte voit plus loin. Un réseau de neurones, le transformer du chapitre 1.4, remplacera les comptes : il rapproche les contextes qui se ressemblent, et peut donc deviner après un contexte jamais vu. Et elle apprendra au chapitre 1.5, sur bien plus de texte que ce site.

Sa mesure, elle, ne changera pas : la perte qu'on suit pendant son entraînement est la moyenne des −ln de l'étape 7, le logarithme de la perplexité. Comme le modèle lissé, son réseau ne donne jamais une chance nulle : softmax, au chapitre 0.2, rend toujours des probabilités plus grandes que zéro.

<EncadreLabo sources={LABO.sources}>

Le modèle de ce chapitre regarde au plus {N_MAX} caractères en arrière, choisit parmi {TEXTE.v} caractères, et a appris sur {TEXTE.entrainement} caractères.

GPT-3, d'OpenAI, s'est entraîné sur des passages de {LABO.gpt3.contexte} jetons, choisit parmi {LABO.gpt3.vocabulaire} jetons, et a appris sur {LABO.gpt3.jetons} de jetons.
