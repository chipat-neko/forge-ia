export const DEFI = (() => {
  // La réponse du défi : avec un message système et un deuxième tour, seule « une pensée » dépasse la moitié de
  // jetons appris (calculé par le jumeau) ; la meilleure des autres en reste loin.
  const tout = (nom) => ({ nom, systeme: true, deuxieme: true });
  const etats = CONVERSATIONS.map((c) => ({ nom: c.nom, etat: etatDeLAtelier(CONVERSATIONS, tout(c.nom)) }));
  const reussis = etats.filter((x) => defiReussi(x.etat, tout(x.nom))).map((x) => x.nom);
  exiger(reussis.length === 1 && reussis[0] === 'pensee', 'avec les deux ajouts, une autre conversation que « une pensée » réussit le défi, ou aucune');
  const pensee = etats.find((x) => x.nom === 'pensee').etat;
  const autres = etats.filter((x) => x.nom !== 'pensee' && x.etat.erreur === null);
  const meilleure = autres.reduce((m, x) => (x.etat.appris / x.etat.jetons > m.etat.appris / m.etat.jetons ? x : m));
  // « Seule, la conversation du bonjour dépasse la moitié ; avec les deux ajouts, elle tombe sous la moitié. »
  const bonjour = etatDeLAtelier(CONVERSATIONS, { nom: 'sans-systeme', systeme: false, deuxieme: false });
  const bonjourTout = etatDeLAtelier(CONVERSATIONS, tout('sans-systeme'));
  exiger(2 * bonjour.appris > bonjour.jetons && 2 * bonjourTout.appris <= bonjourTout.jetons, 'le bonjour seul ne dépasse plus la moitié, ou la dépasse encore avec les deux ajouts');
  // « Sa réponse est plus longue que sa question. »
  const c = CONVERSATIONS.find((x) => x.nom === 'sans-systeme');
  exiger(c.tours[1].ids.length > c.tours[0].ids.length, 'la réponse du bonjour n’est plus plus longue que sa question');
  // « Le message système n'ajoute que des jetons lus » : la même conversation, avec lui, a autant de jetons appris.
  const bonjourSysteme = etatDeLAtelier(CONVERSATIONS, { nom: 'sans-systeme', systeme: true, deuxieme: false });
  exiger(bonjourSysteme.appris === bonjour.appris && bonjourSysteme.jetons > bonjour.jetons, 'un message système ajoute maintenant des jetons appris');
  const part = (e) => pourcent(e.appris / e.jetons, 0);
  const nom = (n) => typographier(`« ${LIBELLES_DE_L_ATELIER[n]} »`);
  return {
    pensee: nom('pensee'),
    bonjour: nom('sans-systeme'),
    meilleure: nom(meilleure.nom),
    appris: nombre(pensee.appris),
    jetons: nombre(pensee.jetons),
    partPensee: part(pensee),
    partMeilleure: part(meilleure.etat),
    partBonjour: part(bonjour),
    partBonjourTout: part(bonjourTout),
  };
})();

export const FIGURE = (() => {
  const f = empaquetagesDeLaFigure(RACINE);
  const { exemples, plusLong } = resumeDesLongueurs(f.longueurs);
  exiger(SFT.longueur === LONGUEUR_DE_TON_IA, 'les rangées de ton IA n’ont plus la longueur que la figure montre à l’arrivée');
  // « Plus les rangées sont longues, moins il en écarte ; à partir de celles de ton IA, aucun. »
  f.resultats.forEach((r, i) => i === 0 || exiger(r.ecartes <= f.resultats[i - 1].ecartes, `des rangées de ${r.longueur} jetons écartent plus que de plus courtes`));
  exiger(f.resultats.filter((r) => r.longueur >= LONGUEUR_DE_TON_IA).every((r) => r.ecartes === 0), 'des rangées de 2 048 jetons écartent maintenant des exemples');
  exiger(f.resultats[0].ecartes > 0, 'des rangées de 512 jetons n’écartent plus aucun exemple');
  // « Avec les rangées de ton IA, BFD remplit presque toutes les places, {part} » (plus de 99 %, borne du texte) ;
  // « avec des rangées plus courtes, … les places se remplissent moins bien ».
  const part = (r) => r.occupees / (r.rangees * r.places);
  const sienne = f.resultats.find((r) => r.longueur === LONGUEUR_DE_TON_IA);
  exiger(part(sienne) > 0.99, 'avec les rangées de ton IA, BFD ne remplit plus presque toutes les places');
  exiger(f.resultats.filter((r) => r.longueur < LONGUEUR_DE_TON_IA).every((r) => part(r) < part(sienne)), 'des rangées plus courtes se remplissent maintenant aussi bien que celles de ton IA');
  // « Dans le jeu de ton IA, la moitié des conversations font au plus {mediane} jetons » : moins de la moitié d'une rangée.
  exiger(f.source === 'jeu', 'la figure ne range plus les longueurs du jeu de ton IA (mesures/longueurs-sft-1.7.json)');
  const triees = [...lireLongueurs(RACINE).longueurs].sort((a, b) => a - b);
  const mediane = triees[Math.floor((triees.length - 1) / 2)];
  exiger(2 * mediane < SFT.longueur, 'la moitié des conversations du jeu ne tient plus dans une demi-rangée');
  return { ...f, exemples, plusLong, mediane: nombre(mediane), remplies: pourcent(part(sienne), 1), longueur: nombre(SFT.longueur), court: nombre(LONGUEURS_DE_RANGEE[0]) };
})();

Il n'a été corrigé que par la sortie, où les poids liés du {LIENS.l14 ? <a href={LIENS.l14}>chapitre 1.4</a> : 'chapitre 1.4'} le font aussi servir : à chaque pas, ton IA apprenait seulement à ne jamais prédire ces jetons. Ce qu'ils veulent dire quand elle les lit, personne ne le lui a appris. C'est une déduction, pas une mesure. Elle pourrait expliquer en partie les réponses de l'étape 1 : la question lui arrive entourée de jetons qu'elle n'a jamais lus. L'affinage, lui, les lui fait lire dans chaque conversation.

<Etape numero={4} titre="Ranger les conversations">

L'affinage, comme le pré-entraînement, lit des rangées de longueur fixe : {FIGURE.longueur} jetons pour ton IA, la longueur de son contexte. Une conversation est souvent bien plus courte : dans le jeu de ton IA, la moitié en font au plus {FIGURE.mediane}. Une seule par rangée, et le reste serait du bourrage : des places perdues, et du calcul pour rien. Forge en range donc plusieurs par rangée, sans jamais en couper une : c'est l'empaquetage. Il suit BFD, de l'anglais *best-fit decreasing* : les plus longues d'abord, chacune dans la rangée où elle laisse le moins de place libre.

<FigureEmpaquetage client:load resultats={FIGURE.resultats} source={FIGURE.source} exemples={FIGURE.exemples} plusLong={FIGURE.plusLong} />
