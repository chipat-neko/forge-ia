export const DEFI = (() => {
  // La réponse du défi : avec un message système et un deuxième tour, seule « une pensée » dépasse la moitié de
  // jetons appris (calculé par le jumeau) ; la meilleure des autres en reste loin.
  const tout = (nom) => ({ nom, systeme: true, deuxieme: true });
  const etats = CONVERSATIONS.map((c) => ({ nom: c.nom, etat: etatDeLAtelier(CONVERSATIONS, tout(c.nom)) }));
  const reussis = etats.filter((x) => defiReussi(x.etat, tout(x.nom))).map((x) => x.nom);
  exiger(reussis.length === 1 && reussis[0] === 'pensee', 'avec les deux ajouts, une autre conversation que « une pensée » réussit le défi, ou aucune');
  const pensee = etats.find((x) => x.nom === 'pensee').etat;
  const autres = etats.filter((x) => x.nom !== 'pensee' && x.etat.erreur === null);
  const meilleure = autres.reduce((m, x) => (x.etat.appris / x.etat.jetons > m.etat.appris / m.etat.jetons ? x : m));
  // « Seule, la conversation du bonjour dépasse la moitié ; avec les deux ajouts, elle tombe sous la moitié. »
  const bonjour = etatDeLAtelier(CONVERSATIONS, { nom: 'sans-systeme', systeme: false, deuxieme: false });
  const bonjourTout = etatDeLAtelier(CONVERSATIONS, tout('sans-systeme'));
  exiger(2 * bonjour.appris > bonjour.jetons && 2 * bonjourTout.appris <= bonjourTout.jetons, 'le bonjour seul ne dépasse plus la moitié, ou la dépasse encore avec les deux ajouts');
  // « Sa réponse est plus longue que sa question. »
  const c = CONVERSATIONS.find((x) => x.nom === 'sans-systeme');
  exiger(c.tours[1].ids.length > c.tours[0].ids.length, 'la réponse du bonjour n’est plus plus longue que sa question');
  // « Le message système n'ajoute que des jetons lus » : la même conversation, avec lui, a autant de jetons appris.
  const bonjourSysteme = etatDeLAtelier(CONVERSATIONS, { nom: 'sans-systeme', systeme: true, deuxieme: false });
  exiger(bonjourSysteme.appris === bonjour.appris && bonjourSysteme.jetons > bonjour.jetons, 'un message système ajoute maintenant des jetons appris');
  const part = (e) => pourcent(e.appris / e.jetons, 0);
  const nom = (n) => typographier(`« ${LIBELLES_DE_L_ATELIER[n]} »`);
  return {
    pensee: nom('pensee'),
    bonjour: nom('sans-systeme'),
    meilleure: nom(meilleure.nom),
    appris: nombre(pensee.appris),
    jetons: nombre(pensee.jetons),
    partPensee: part(pensee),
    partMeilleure: part(meilleure.etat),
    partBonjour: part(bonjour),
    partBonjourTout: part(bonjourTout),
  };
})();

<Etape numero={4} titre="Ranger les conversations">

L'affinage, comme le pré-entraînement, lit des rangées de longueur fixe : {FIGURE.longueur} jetons pour ton IA, la longueur de son contexte. Une conversation est souvent bien plus courte. Une seule par rangée, et le reste serait du bourrage : des places perdues, et du calcul pour rien. Forge en range donc plusieurs par rangée, sans jamais en couper une : c'est l'empaquetage. Il suit BFD, de l'anglais *best-fit decreasing* : les plus longues d'abord, chacune dans la rangée où elle laisse le moins de place libre.

<FigureEmpaquetage client:load resultats={FIGURE.resultats} source={FIGURE.source} exemples={FIGURE.exemples} plusLong={FIGURE.plusLong} />

Avec des rangées assez longues pour les plus longues conversations, aucune n'est écartée, et BFD remplit presque toutes les places. Avec des rangées plus courtes, de {FIGURE.court} jetons par exemple, il faut écarter les conversations trop longues : forge n'en coupe jamais une, car sa fin serait perdue. Voici l'empaquetage de forge :
