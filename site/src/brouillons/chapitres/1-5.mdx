export const DEFI = (() => {
  // Le défi de l'atelier : ses pertes viennent du moteur du mini-modèle (defi-1-5.ts).
  const d = resultatDuDefi();
  const ecart = (x) => nombre(x - d.defaut.fin, 3);
  // « Trois fois le taux habituel … dix fois, non » : des rapports entiers.
  const fois = (taux) => {
    const r = taux / d.defaut.taux;
    exiger(Math.abs(r - Math.round(r)) < 1e-9 * r, `le taux ${taux} n’est plus un multiple entier du taux habituel`);
    return Math.round(r);
  };
  const enLettres = (n) => ({ 3: 'trois', 10: 'dix', 30: 'trente', 100: 'cent' })[n] ?? nombre(n);
  exiger(Number.isFinite(d.suivant.fin), 'le taux qui suit la réponse du défi explose : son explication est à revoir');
  exiger(d.suivant.fin > d.defaut.fin, 'le taux qui suit la réponse du défi finit plus bas que le défaut');
  return {
    ecartMax: nombre(ECART_DU_DEFI, 2),
    defaut: reglage(d.defaut.taux),
    finDefaut: formaterPerte(d.defaut.fin),
    reponse: reglage(d.reponse.taux),
    finReponse: formaterPerte(d.reponse.fin),
    ecartReponse: ecart(d.reponse.fin),
    plusBas: d.reponse.fin < d.defaut.fin,
    suivant: reglage(d.suivant.taux),
    finSuivant: formaterPerte(d.suivant.fin),
    ecartSuivant: ecart(d.suivant.fin),
    // En tête de phrase : une capitale.
    foisReponse: enLettres(fois(d.reponse.taux)).replace(/^./, (c) => c.toUpperCase()),
    foisSuivant: enLettres(fois(d.suivant.taux)),
  };
})();

export const COURTE = (() => {
  // Le pari « courte » et le résumé de la version d'une heure (son journal, relevé par le projet).
  const r = MINI.resume;
  const f = faitsDuPariDebit(MINI);
  const mesure = MESURES_TON_IA.mini.carte.jetonsParSeconde;
  const ecartMesure = Math.abs(f.debit / mesure - 1);
  exiger(ecartMesure < 0.01, 'le débit de l’heure entière s’écarte de 1 % ou plus de celui mesuré avant');
  // L'acquis : « des centaines de milliers de jetons chaque seconde ».
  exiger(f.debit >= 1e5 && f.debit < 1e6, 'le débit de la version d’une heure n’est plus de quelques centaines de milliers de jetons par seconde');
  exiger(r.pauses === 0 && MINI.sessions.length === 1, 'la version d’une heure a maintenant des pauses : son texte est à revoir');
  // La première validation, au pas evaluation_tous_pas : déjà plus de la moitié du chemin.
  const [pasVal1, val1] = MINI.courbe.validations[0];
  exiger(pasVal1 === REGLAGES.brut.evaluation_tous_pas, 'la première validation n’est plus au premier palier de validation');
  const chemin = (r.perte_premier_pas - val1) / (r.perte_premier_pas - r.validation_finale);
  exiger(chemin > 0.5, 'à la première validation, la perte n’a plus fait la moitié du chemin');
  exiger(r.validation_finale > r.perte_finale, 'à la fin, la perte de validation n’est plus au-dessus de celle d’entraînement');
  // « Un peu plus haut » : à moins d'un demi-point.
  exiger(r.validation_finale - r.perte_finale < 0.5, 'à la fin, la perte de validation n’est plus « un peu » au-dessus de celle d’entraînement');
  return {
    recentes: nombre(lirePertesRecentes(RACINE)),
    choix: choixDebitCourte(MINI),
    carte: materiel(MATERIEL_DES_MESURES).nom,
    debit: nombre(f.debit),
    duree: dureeDeCalcul(r.secondes_de_calcul),
    jetons: nombre(r.jetons_vus),
    pas: nombre(r.pas_faits),
    lot: nombre(f.lot),
    parJeton: nombre(f.jetonsParParametre),
    parametres: nombre(r.parametres),
    mesure: nombre(mesure),
    ecartMesure: pourcent(ecartMesure, 1),
    premier: formaterPerte(r.perte_premier_pas),
    finale: formaterPerte(r.perte_finale),
    validation: formaterPerte(r.validation_finale),
    pasVal1: nombre(pasVal1),
    val1: formaterPerte(val1),
    partDesPas: pourcent(pasVal1 / r.pas_total, 1),
  };
})();

<Etape numero={1} titre="Ton IA naît">

<OuvertureNaissance />

Voici ton IA dans sa version d'une heure, la taille « {REGLAGES.taille} » : {REGLAGES.parametres} paramètres, le transformer du {LIENS.l14 ? <a href={LIENS.l14}>chapitre 1.4</a> : 'chapitre 1.4'} dans sa plus petite taille. Au premier pas, ils ne savent rien ; au dernier, le {REGLAGES.pasTotal}ᵉ, ils ont été corrigés à chaque pas, d'après tout ce qu'elle a lu.

Entre les deux, le pré-entraînement : elle lit des textes, morceau par morceau, et apprend à prédire le jeton qui vient ensuite. C'est son premier entraînement, et le plus long. Ce chapitre l'ouvre pièce par pièce : la perte, le lot, le taux, l'optimiseur, puis la pause et la reprise. Un petit modèle refait la même recette dans ta page, et la version d'une heure se refait sur ta carte graphique. Ton IA, la vraie, suit le même chemin, en bien plus long.

</Etape>

<Etape numero={2} titre="La perte au départ">

À chaque position d'un texte, ton IA donne une probabilité à chacun des {DEPART.vocabulaire} jetons de son vocabulaire, celui du {LIENS.l13 ? <a href={LIENS.l13}>chapitre 1.3</a> : 'chapitre 1.3'}. Sa perte, c'est l'entropie croisée du {LIENS.l02 ? <a href={LIENS.l02}>chapitre 0.2</a> : 'chapitre 0.2'} : <span class="formule-en-texte">−ln p</span>, où p est la probabilité qu'elle donnait au bon jeton, en moyenne sur tous les jetons d'un lot. Elle vaut 0 si ton IA était sûre du bon jeton, et grandit à mesure qu'elle lui en donnait moins.

<Pari
  id="depart"
  question="Avant d'avoir rien appris, quelle perte a ton IA au premier pas ?"
  precision={`Ses poids sont tirés au hasard. Elle choisit parmi ${DEPART.vocabulaire} jetons.`}
  choix={choixPerteDepartTonIA(MINI)}
  acquis="depart"
  ouvre="depart"
>

Plus de 5 : au premier pas de la version d'une heure, elle valait {DEPART.premier}. Avec des poids tirés au hasard, ton IA donne à peu près la même chance à chacun des {DEPART.vocabulaire} jetons : p vaut à peu près 1 sur {DEPART.vocabulaire}, et la perte, <span class="formule-en-texte">ln {DEPART.vocabulaire}</span>, soit {DEPART.hasard}. C'est la perte du hasard : ln V, pour un vocabulaire de V jetons. Celle du premier pas n'en est éloignée que de {DEPART.ecart} : au départ, ton IA ne sait vraiment rien. Cette perte est maintenant dévoilée dans l'ouverture, au pas 1.

Tout l'entraînement consiste à faire descendre ce nombre. Il ne descendra jamais à 0 : même un texte bien connu laisse plusieurs suites possibles.

</Pari>

</Etape>

<Etape numero={3} titre="Le lot, le taux, le planning">

Un pas d'entraînement, c'est la boucle du {LIENS.l05 ? <a href={LIENS.l05}>chapitre 0.5</a> : 'chapitre 0.5'} : lire un lot, mesurer la perte, calculer le gradient, corriger les poids. À chaque pas, la version d'une heure lit un lot de {REGLAGES.lot} jetons : {REGLAGES.lignes} lignes de {REGLAGES.contexte} jetons, la longueur de son contexte. Pour chaque ligne, elle doit deviner, à chaque position, le jeton qui suit.

La carte graphique ne traite pas ces {REGLAGES.lignes} lignes d'un coup. Elle les prend par micro-lots de {REGLAGES.microLot} lignes, le réglage le plus rapide mesuré sur la carte de l'auteur, et additionne leurs gradients avant de toucher aux poids : {REGLAGES.accumulation} micro-lots par pas. C'est l'accumulation de gradient. Le pas est le même que si la carte avait tout lu d'un coup.

Reste à savoir de combien corriger les poids : c'est le taux d'apprentissage. Ton IA ne le garde pas fixe. Elle suit un planning, celui de forge :

<CodeExplique fichier="projets/forge/forge/entrainement/planning.py" symbole="taux">
  <li data-lignes="1">`taux` reçoit le numéro du pas, le nombre de pas, le taux maximal et la durée de l'échauffement. `ratio_min` dit où finit la descente : à {REGLAGES.ratio} fois le taux maximal.</li>
  <li data-lignes="3-4">L'échauffement : le taux monte en ligne droite, de 0 au taux maximal. Au début, les poids sont tirés au hasard et les gradients partent dans tous les sens : de grands pas déstabiliseraient le modèle. Pour la version d'une heure, au pas {REGLAGES.demiEchauffement}, à mi-échauffement, il vaut {REGLAGES.demiTaux}.</li>
  <li data-lignes="5-6">Après le dernier pas, le taux ne bouge plus.</li>
  <li data-lignes="7">`avancement` va de 0, à la fin de l'échauffement, à 1, au dernier pas.</li>
  <li data-lignes="8">La descente suit une demi-vague de cosinus : cos 0 vaut 1, et le taux est au maximum ; cos π vaut −1, et il est au minimum. Elle est douce au début et à la fin, plus rapide au milieu.</li>
</CodeExplique>

Pour la version d'une heure : le taux monte de 0 à {REGLAGES.tauxMax} pendant {REGLAGES.echauffement} pas, puis redescend jusqu'à {REGLAGES.tauxMin} au pas {REGLAGES.pasTotal}. En fin d'entraînement, de petits pas affinent ce qui est appris sans le défaire.

Le taux ne dépend que du numéro du pas. Après une pause, il repart donc au bon endroit sans rien garder d'autre que ce numéro : l'étape 6 s'en servira.

</Etape>

<Etape numero={5} titre="AdamW et l'écrêtage">

La descente de gradient du {LIENS.l02 ? <a href={LIENS.l02}>chapitre 0.2</a> : 'chapitre 0.2'} recule chaque poids du taux fois sa pente. C'est l'optimiseur le plus simple : la recette qui corrige les poids à chaque pas. Ton IA, elle, utilise AdamW : pour chaque poids, il garde deux moyennes, celle de ses pentes récentes et celle de leurs carrés. Il avance d'après la première, divisée par la racine de la seconde. Un poids dont la pente garde le même signe avance d'à peu près le taux à chaque pas ; un poids dont la pente change sans cesse de signe avance moins. Voici comment forge le crée :

<CodeExplique fichier="projets/forge/forge/entrainement/boucle.py" symbole="_creer_optimiseur">
  <li data-lignes="1">`_creer_optimiseur` reçoit le modèle, la configuration de l'entraînement et l'appareil.</li>
  <li data-lignes="2-3">Les poids sont rangés en deux familles : les matrices, qui ont au moins deux dimensions (la table du plongement, les projections de l'attention et du MLP), et les vecteurs, les poids des RMSNorm.</li>
  <li data-lignes="4-7">La décroissance des poids, {REGLAGES.decroissance} pour la version d'une heure, ne s'applique qu'aux matrices : à chaque pas, elle tire doucement chacun de leurs poids vers 0.</li>
  <li data-lignes="8-9">AdamW, avec le taux maximal et `betas`, {REGLAGES.beta1} et {REGLAGES.beta2} : la mémoire des deux moyennes. Plus un beta est proche de 1, plus sa moyenne se souvient loin. `fused` : sur la carte, une seule opération met à jour tous les poids.</li>
</CodeExplique>

Un lot bizarre peut donner un gradient énorme, et un saut qui défait des heures d'apprentissage. Avant chaque pas d'AdamW, forge l'écrête : si la norme du gradient, la longueur de toutes ses pentes mises bout à bout, dépasse {REGLAGES.ecretage}, toutes les pentes sont réduites dans la même proportion. Voici un pas entier :

<CodeExplique fichier="projets/forge/forge/entrainement/boucle.py" symbole="_pas_d_optimisation">
  <li data-lignes="1-9">Un pas reçoit le modèle, l'optimiseur, le chargeur des jetons, le nombre de micro-lots par pas, le taux de ce pas (celui du planning), le seuil de l'écrêtage et l'appareil.</li>
  <li data-lignes="10-11">Le taux du planning est donné à l'optimiseur.</li>
  <li data-lignes="12-13">Les gradients du pas d'avant sont effacés, et la somme des pertes part de 0.</li>
  <li data-lignes="14-16">Un micro-lot après l'autre : `x`, des lignes de jetons, et `y`, les mêmes décalées d'un jeton, les jetons à deviner.</li>
  <li data-lignes="17-18">Le modèle calcule sa perte. Sur la carte, `_autocast` fait les calculs en bfloat16, des nombres sur 16 bits, plus rapides.</li>
  <li data-lignes="19-20">Chaque perte est divisée par le nombre de micro-lots avant `backward` : les gradients s'additionnent, et leur somme est celle de la perte moyenne du lot entier. `somme` garde les pertes, pour le journal.</li>
  <li data-lignes="21">L'écrêtage : `clip_grad_norm_` réduit les gradients si leur norme dépasse le seuil, et rend la norme d'avant.</li>
  <li data-lignes="22-24">La perte moyenne du pas. Si elle, ou la norme, n'est plus un nombre fini, tout s'arrête avant de toucher aux poids : l'entraînement a divergé.</li>
  <li data-lignes="25-26">AdamW corrige les poids, et le pas rend sa perte, celle que note le journal.</li>
</CodeExplique>

</Etape>

<Etape numero={6} titre="Pause et reprise">

La version d'une heure tient en une heure. Ton IA, la vraie, demande des jours de calcul, sur un PC qui sert aussi à autre chose. Il faut pouvoir l'arrêter, puis la reprendre plus tard.

<Pari
  id="reprise"
  question="Après une pause, la perte du pas repris…"
  precision="Un entraînement est arrêté à mi-chemin, sauvegardé, puis repris le lendemain."
  choix={ESSAI.choix}
  acquis="reprise"
>

Elle continue la courbe, au même nombre près. Le projet du chapitre le vérifie : un petit modèle s'entraîne {ESSAI.pas} pas d'un trait, puis de nouveau, arrêté au pas {ESSAI.pause} et repris. Les deux finissent sur la même perte, {ESSAI.perte}, au bit près, et chaque pas repris a exactement la perte qu'il avait d'un trait.

Pour cela, la sauvegarde garde tout ce dont le pas suivant dépend :

- les poids du modèle ;
- l'état d'AdamW : les deux moyennes de chaque poids ;
- le numéro du pas, qui suffit au planning du taux ;
- la position dans les données : le prochain morceau de texte à lire ;
- l'état du hasard, pour qu'il tire les mêmes nombres qu'avant ;
- et la configuration, pour refuser une reprise qui changerait le calcul.

Sans les moyennes d'AdamW, les pas suivants ne seraient plus les mêmes ; sans la position dans les données, ton IA relirait des textes déjà lus.

</Pari>

Voici le test du projet qui le vérifie, sur le processeur :

<CodeExplique fichier="projets/1.5-pre-entrainement/tests/test_essai.py" region="test-reprise">
  <li data-lignes="2">`essai_pause_reprise` entraîne le petit modèle trois fois : d'un trait, puis arrêté à mi-chemin, puis repris jusqu'au bout.</li>
  <li data-lignes="3">L'écart des deux pertes finales doit être exactement 0, pas seulement petit.</li>
  <li data-lignes="4-5">Et la seconde moitié, refaite pas à pas, a les mêmes pertes que d'un trait.</li>
</CodeExplique>

Et la commande de l'essai, avec ce qu'elle a affiché :

<SortieCommande
  fichier={MESURES_PARTIE_1.sortieEssaiProcesseur}
  commande={PRE_ENTRAINEMENT_ESSAI}
  legende="L'essai de pause et de reprise, sur le processeur"
  absent="À mesurer : l'essai de pause et de reprise sur le processeur."

Avec forge, c'est la même chose, en grand. Pendant l'entraînement de la version d'une heure, que l'étape suivante te fait lancer, {PAUSE} : le pas en cours se termine, tout est sauvegardé, puis forge affiche la commande de reprise et s'arrête. Pour reprendre exactement là :

</Fragment>

<Etape numero={9} titre="Chez un grand labo">

Ton IA s'entraîne sur une seule carte. Les grands labos en font travailler des milliers ensemble.

<EncadreLabo sources={SOURCES_LABO_1_5}>

Llama 3.1 405B, de Meta, a demandé {LABO.heures} d'heures de calcul de cartes H100, des cartes de centre de données bien plus puissantes que celle de l'auteur. La version d'une heure de ton IA a pris {LABO.heuresMini} de calcul, sur une seule carte : environ {LABO.foisHeures} de fois moins. Meta l'a entraîné avec jusqu'à {LABO.cartes} cartes à la fois. Son pré-entraînement a compté {LABO.operations} opérations, environ {LABO.foisOps} de fois les {LABO.opsMini} de la version d'une heure, comptées comme le calculateur : {LABO.parJeton} par paramètre et par jeton lu.

Avec autant de cartes, les pannes sont ordinaires. Sur une période de {LABO.jours} jours de son pré-entraînement, Meta a compté {LABO.interruptions} interruptions : {LABO.prevues} prévues, pour de la maintenance automatique ou des opérations lancées par l'équipe, et {LABO.imprevues} imprévues.
